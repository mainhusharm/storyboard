    <script>
    (function() {
        const overlay    = document.getElementById('ssh-overlay');
        const output     = document.getElementById('ssh-output');
        const cursorLine = document.getElementById('ssh-cursor-line');
        const skipBtn    = document.getElementById('ssh-skip-btn');
        const statusText = document.getElementById('ssh-status-text');

        let skipped = false;
        let swaggerMeta = null;
        let animResolve = null;
        let swaggerResolve = null;
        let swaggerShown = false;

        const sleep = ms => new Promise(r => setTimeout(r, ms));

        function addLine(text, cls) {
            const d = document.createElement('div');
            d.className = 'ssh-line' + (cls ? ' ' + cls : '');
            d.textContent = text;
            output.appendChild(d);
            output.scrollTop = output.scrollHeight;
            return d;
        }

        async function typeLine(text, speed, cls) {
            speed = speed || 38;
            const d = addLine('', cls);
            for (const ch of text) {
                if (skipped) { d.textContent = text; break; }
                d.textContent += ch;
                output.scrollTop = output.scrollHeight;
                await sleep(speed + Math.random() * 14 - 7);
            }
            return d;
        }

        async function pause(ms) {
            if (!skipped) await sleep(ms);
        }

        async function waitForSwagger() {
            if (swaggerMeta) return;
            await new Promise(r => { swaggerResolve = r; });
        }

        async function showSwaggerLines() {
            if (swaggerShown) return;
            swaggerShown = true;
            if (!swaggerMeta) return;
            const { totalEndpoints, totalCategories, description } = swaggerMeta;
            const shortDesc = (description || 'REST API services for bots and innovative projects').split('\n')[0].slice(0, 68);
            await typeLine('[server]$ registry: ' + totalEndpoints + ' endpoints across ' + totalCategories + ' categories', 26, 'ssh-ok');
            await pause(80);
            addLine('[server]$ info: ' + shortDesc, 'ssh-dim');
        }

        async function runSequence() {
            await pause(200);

            cursorLine.textContent = '[user@local ~]$ ';
            await pause(280);
            await typeLine('[user@local ~]$ ssh paxsenix@api.paxsenix.org -p 22', 40, 'ssh-prompt');
            await pause(300);
            addLine('The authenticity of host \'api.paxsenix.org\' can\'t be established.', 'ssh-dim');
            addLine('ED25519 key fingerprint is SHA256:p4xS3n1xAP1/d0c5+k3yFiNg3rPr1NT==', 'ssh-dim');
            await pause(250);
            await typeLine('Are you sure you want to continue connecting (yes/no)? yes', 38, 'ssh-prompt');
            await pause(200);
            addLine('Warning: Permanently added \'api.paxsenix.org\' to the list of known hosts.', 'ssh-warn');
            addLine('');
            await pause(260);

            const banner = [
                ' █████╗ ██████╗   ██╗',
                '██╔══██╗██╔══██╗██║',
                '███████║██████╔╝██║',
                '██╔══██║██╔═══╝  ██║',
                '██║   ██║██║        ██║',
                '╚═╝   ╚═╝╚═╝        ╚═╝',
            ];
            for (const row of banner) {
                addLine(row, 'ssh-banner');
                await pause(skipped ? 0 : 50);
            }
            addLine('');
            addLine('  PaxSenix API Server  ·  Documentation v3.3.4  ·  2026', 'ssh-info');
            addLine('');
            await pause(220);

            statusText.textContent = 'Loading registry...';
            cursorLine.textContent = '[server]$ ';
            await typeLine('[server]$ source /etc/api-registry --init --verbose', 36, 'ssh-prompt');
            await pause(280);

            if (swaggerMeta) {
                await showSwaggerLines();
            } else {
                addLine('[server]$ Waiting for registry sync...', 'ssh-dim');
                await waitForSwagger();
                await showSwaggerLines();
            }

            await pause(160);
            await typeLine('[server]$ boot docs-ui --env=production', 34, 'ssh-prompt');
            await pause(220);
            addLine('[server]$ Compiling routes...                 [  OK  ]', 'ssh-ok');
            await pause(120);
            addLine('[server]$ Mounting endpoint explorer...       [  OK  ]', 'ssh-ok');
            await pause(120);
            addLine('[server]$ Starting interactive interface...   [  OK  ]', 'ssh-ok');
            await pause(180);
            addLine('');
            addLine('  ✓ Session established. Welcome, developer.', 'ssh-success');

            statusText.textContent = 'Connected';
            cursorLine.textContent = '[paxsenix@api ~]$ ';
            await pause(700);
            if (!skipped) finishReveal();
        }

        window.addEventListener('swaggerLoaded', function(e) {
            var d = e.detail;
            swaggerMeta = {
                totalEndpoints: d.apiData.categories.reduce(function(a, c) { return a + c.endpoints.length; }, 0),
                totalCategories: d.apiData.categories.length,
                description: (d.swaggerDoc.info && d.swaggerDoc.info.description) || ''
            };
            if (swaggerResolve) { swaggerResolve(); swaggerResolve = null; }
        }, { once: true });

        function finishReveal() {
            overlay.classList.add('ssh-exit');
            setTimeout(function() {
                overlay.style.display = 'none';
                document.body.classList.remove('pre-reveal');
                document.body.classList.add('revealed');
            }, 520);
        }

        skipBtn.addEventListener('click', function() {
            skipped = true;
            if (swaggerResolve) { swaggerResolve(); swaggerResolve = null; }
            finishReveal();
        });

        runSequence();
    })();
    </script>

    <script>
    (function() {
        var scrollEl, leftBtn, rightBtn;
        var STEP = 200;

        function updateBtns() {
            if (!scrollEl) return;
            var atStart = scrollEl.scrollLeft <= 4;
            var atEnd = scrollEl.scrollLeft + scrollEl.clientWidth >= scrollEl.scrollWidth - 4;
            leftBtn.classList.toggle('cat-btn-hidden', atStart);
            rightBtn.classList.toggle('cat-btn-hidden', atEnd && scrollEl.scrollWidth <= scrollEl.clientWidth + 8);
        }

        document.addEventListener('DOMContentLoaded', function() {
            scrollEl = document.getElementById('cat-tabs-scroll');
            leftBtn = document.getElementById('cat-scroll-left');
            rightBtn = document.getElementById('cat-scroll-right');
            if (!scrollEl) return;

            leftBtn.addEventListener('click',  function() { scrollEl.scrollBy({ left: -STEP, behavior: 'smooth' }); });
            rightBtn.addEventListener('click', function() { scrollEl.scrollBy({ left: STEP,  behavior: 'smooth' }); });
            scrollEl.addEventListener('scroll', updateBtns, { passive: true });

            var chips = document.getElementById('category-chips');
            if (chips) {
                var mo = new MutationObserver(function() { setTimeout(updateBtns, 50); });
                mo.observe(chips, { childList: true });
            }
            updateBtns();
        });
    })();
    </script>

    <script>
    (function() {
        var panel = document.getElementById('log-panel');
        var entries = document.getElementById('log-entries');
        var badge = document.getElementById('log-badge');
        var header = document.getElementById('log-header');
        var toggle = document.getElementById('log-toggle');
        var count = 0;
        var collapsed = true;

        function togglePanel() {
            collapsed = !collapsed;
            panel.classList.toggle('log-collapsed', collapsed);
            toggle.querySelector('.mso').textContent = collapsed ? 'expand_less' : 'expand_more';
            if (!collapsed) { badge.classList.add('hidden'); }
        }

        header.addEventListener('click', togglePanel);
        panel.classList.add('log-collapsed');

        window.addEventListener('apiRequest', function(e) {
            var d = e.detail;
            count++;
            var cls = d.status < 300 ? 'log-ok' : d.status < 400 ? 'log-warn' : 'log-err';
            var row = document.createElement('div');
            row.className = 'log-entry';
            row.innerHTML =
                '<span class="log-method log-method-' + d.method.toLowerCase() + '">' + d.method + '</span>' +
                '<span class="log-path">' + d.path + '</span>' +
                '<span class="log-status ' + cls + '">' + d.status + '</span>' +
                '<span class="log-dur">' + d.duration + 'ms</span>';
            entries.appendChild(row);
            entries.scrollTop = entries.scrollHeight;

            panel.classList.remove('log-hidden');
            if (collapsed) {
                badge.textContent = count;
                badge.classList.remove('hidden');
                panel.classList.add('log-ping');
                setTimeout(function() { panel.classList.remove('log-ping'); }, 600);
            }
        });
    })();
    </script>
